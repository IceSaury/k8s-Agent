# Agent framework - exact versions from the spec PDF (section 2.1)
langchain==1.2.10
langgraph==1.0.9
langchain-mcp-adapters==0.2.1

# MCP SDK upper bound is required: the adapters declare only "mcp>=1.9.2", so pip
# resolves mcp 2.x, which dropped mcp.shared.context.RequestContext and breaks
# "from langchain_mcp_adapters.client import MultiServerMCPClient".
mcp<2

# langgraph-prebuilt must be pinned low: langchain 1.2.10 caps langgraph<1.1.0,
# but prebuilt 1.0.9+ calls langgraph.runtime.ExecutionInfo (added in langgraph
# 1.1), so pip's default of 1.0.13 breaks "from langchain.agents import create_agent".
langgraph-prebuilt==1.0.8

# LLM provider - qwen-plus via DashScope OpenAI-compatible endpoint
langchain-openai

# Search tools. SEARCH_PROVIDER picks one at runtime: duckduckgo (default) / tavily.
# ddgs is the renamed, maintained successor of "duckduckgo-search", and is also what
# langchain-community 0.4.2 imports - pinning the old name breaks.
# tavily stays installed as the documented fallback provider.
langchain-community
ddgs
langchain-tavily

# CLI rendering and config loading
rich
python-dotenv

# YAML parsing/serializing in agent.py: the `resource list` payload comes back as YAML
# and is projected down to an ops view before reaching the model. It was already present
# as a transitive dependency, but we import it directly so it must be declared.
PyYAML

# Feishu bot (stage 6). Feishu official SDK
lark-oapi
